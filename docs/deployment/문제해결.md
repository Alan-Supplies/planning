CREATE TABLE `payments` (
  `id` int unsigned NOT NULL AUTO_INCREMENT,
  `store_id` int NOT NULL COMMENT '매장 ID',
  `order_id` int unsigned NOT NULL COMMENT 'customer_order ID',
  `ordered_at` datetime DEFAULT NULL COMMENT '주문 시간(DEPRECATED)',
  `transacted_at` datetime NOT NULL COMMENT '거래 발생 시점 (PAYMENT: 주문시간, REFUND: 환불시간). 매출 집계 기준 시간',
  `parent_id` int unsigned DEFAULT NULL COMMENT '부모 결제 ID (환불/취소의 경우 원본 결제 ID)',
  `transaction_type` varchar(20) NOT NULL COMMENT '거래 유형 (PAYMENT/REFUND/CANCEL_REFUND)',
  `amount` int DEFAULT NULL COMMENT '결제 금액',
  `method` varchar(20) DEFAULT NULL COMMENT '결제 수단 (CARD/KAKAO_PAY/CASH)',
  `card_company` varchar(20) DEFAULT NULL COMMENT '카드사',
  `card_number4` varchar(4) DEFAULT NULL COMMENT '카드번호 뒷 4자리',
  `approved_number` varchar(20) DEFAULT NULL COMMENT '승인번호',
  `installment` varchar(2) DEFAULT NULL COMMENT '할부 개월수',
  `approved_at` datetime DEFAULT NULL COMMENT '승인 시간',
  `status` varchar(20) NOT NULL DEFAULT 'SUCCESS' COMMENT '결제 상태 (SUCCESS/FAILED/PENDING)',
  `platform` varchar(30) NOT NULL COMMENT '플랫폼',
  `service_type` varchar(20) NOT NULL COMMENT '서비스 타입',
  `device` varchar(20) NOT NULL COMMENT '디바이스',
  `description` text COMMENT '설명',
  `created_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) COMMENT '생성일시',
  `updated_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6) COMMENT '수정일시',
  PRIMARY KEY (`id`),
  KEY `idx_status` (`status`),
  KEY `idx_store_approved` (`store_id`,`approved_at`),
  KEY `idx_payments_store_ordered` (`store_id`,`ordered_at`,`status`,`service_type`,`amount`),
  KEY `idx_store_transacted` (`store_id`,`transacted_at`)
) ENGINE=InnoDB AUTO_INCREMENT=762029 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
